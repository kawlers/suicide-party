/* ========================================= RANDOM SESSION ID ========================================= */ function createSessionID() { const characters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"; let output = ""; for ( let group = 0; group < 3; group++ ) { for ( let i = 0; i < 4; i++ ) { output += characters[ Math.floor( Math.random() * characters.length ) ]; } if ( group < 2 ) { output += "-"; } } document.getElementById( "sessionID" ).textContent = output; }
